class Profile: def __init__(self, name, nickname, gender, attributes, likes, dislikes): self.name = name self.nickname = nickname self.gender = gender self.attributes = attributes self.likes = likes self.dislikes = dislikes def introduce(self): print(f"✨ Hello! Welcome to my profile ✨") print("=" * 40) print(f"👤 Name : {self.name} ({self.nickname})") print(f"✨ Status : {', '.join(self.attributes)}") print("-" * 40) print("💖 Things I Love:") for item in self.likes: print(f" • {item}") print("-" * 40) print("❌ Things I Dislike:") for item in self.dislikes: print(f" • {item}") print("=" * 40) # สร้างโปรไฟล์ my_profile = Profile( name="ปราณปรียา สาลีผล", nickname="ปราณ", gender="Female", attributes=["สวยสุดๆ ✨", "มั่นใจ 💃", "มีสไตล์ 💅"], likes=[ "ความใส่ใจและรักจริง 💖", "การดูแลตัวเองและแฟชั่น 💄", "บรรยากาศสบายๆ และเพลงโปรด 🎵" ], dislikes=[ "เรื่องผู้หญิงและความไม่ชัดเจน 🚫", "คนโกหกหรือไม่รักษาสัญญา 🙅‍♀️", "ความสัมพันธ์ที่ toxic 💔" ] ) # แสดงผลการแนะนำตัว my_profile.introduce()
